cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/../cm_input_check/boards")
set(PICO_BOARD t3p_waveshare_usb_cm CACHE STRING "Exact CM board")
include("${CMAKE_CURRENT_LIST_DIR}/../../local/pico-sdk/pico_sdk_init.cmake")
project(t3p_cm_rom_probe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../local/Pico-PIO-USB" pico_pio_usb)
add_executable(t3p_cm_rom_probe main.c usb_descriptors.c
    ${PICO_TINYUSB_PATH}/src/portable/raspberrypi/pio_usb/hcd_pio_usb.c)
pico_generate_pio_header(t3p_cm_rom_probe ${CMAKE_CURRENT_LIST_DIR}/../cm_wave_scope/key_wave.pio)
target_include_directories(t3p_cm_rom_probe PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(t3p_cm_rom_probe PRIVATE PIO_USB_USE_TINYUSB)
target_compile_options(t3p_cm_rom_probe PRIVATE -Wall -Wextra)
target_link_libraries(t3p_cm_rom_probe pico_stdlib pico_multicore pico_util
    hardware_dma hardware_pio pico_pio_usb tinyusb_device tinyusb_host)
pico_enable_stdio_uart(t3p_cm_rom_probe 0)
pico_enable_stdio_usb(t3p_cm_rom_probe 0)
pico_set_program_name(t3p_cm_rom_probe "CM one attach ROM descriptor probe")
pico_set_program_version(t3p_cm_rom_probe "0.3")
pico_add_extra_outputs(t3p_cm_rom_probe)
